// InputAgentMover.cs
// Namespace: {NAMESPACE}.Agent
// IAgentMover implementation for player-controlled movement via CharacterController.

using UnityEngine;

namespace {NAMESPACE}.Agent
{
    public class InputAgentMover : IAgentMover
    {
        private readonly CharacterController _cc;
        private readonly float               _speed;
        private          Vector3             _lastVelocity;

        public InputAgentMover(CharacterController cc, float speed)
        {
            _cc    = cc;
            _speed = speed;
        }

        public bool IsMoving => _lastVelocity.sqrMagnitude > 0.0001f;

        public void Move(Vector3 velocity)
        {
            _lastVelocity = velocity;
            _cc.Move(velocity * (_speed * Time.deltaTime));
        }

        public void RotateTo(Vector3 targetPosition)
        {
            var dir = (targetPosition - _cc.transform.position);
            dir.y = 0f;
            if (dir.sqrMagnitude < 0.0001f) return;
            _cc.transform.rotation = Quaternion.LookRotation(dir.normalized);
        }
    }
}
